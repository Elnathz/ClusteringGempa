# --- Di bagian atas file (imports) ---
from datetime import datetime

# --- Di section baru atau sidebar ---
st.sidebar.markdown("---")
st.sidebar.subheader("🛡️ Analisis Keselamatan Lokasi")

# 1. Input Lokasi
WILAYAH_BOUNDS = {
    "Sumatra":      {"lat": (-6, 6), "lon": (95, 106)},
    "Jawa":         {"lat": (-9, -5), "lon": (105, 115)},
    "Bali-NTB":     {"lat": (-10, -6), "lon": (115, 120)},
    "Sulawesi":     {"lat": (-6, 2), "lon": (119, 126)},
    "Maluku":       {"lat": (-8, 2), "lon": (124, 136)},
    "Papua":        {"lat": (-9, 0), "lon": (130, 141)},
}
selected_region = st.sidebar.selectbox("Pilih Wilayah", list(WILAYAH_BOUNDS.keys()))

# 2. Deteksi Bulan Real-time
now = datetime.now()
current_month = now.month
current_month_name = now.strftime("%B")  # e.g., "January"
st.sidebar.info(f"🗓️ Bulan Aktif: {current_month_name} {now.year}")

# 3. Filter Data
bounds = WILAYAH_BOUNDS[selected_region]
df_region = df[
    (df['latitude'].between(*bounds['lat'])) & 
    (df['longitude'].between(*bounds['lon']))
]
df_region['month'] = pd.to_datetime(df_region['time']).dt.month
df_month = df_region[df_region['month'] == current_month]

# 4. Hitung Statistik
total = len(df_month)
if total > 0:
    avg_mag = df_month['mag'].mean()
    max_mag = df_month['mag'].max()
    high_risk_count = len(df_month[df_month['cluster'] >= 2])
    pct_high = high_risk_count / total
else:
    avg_mag, max_mag, pct_high = 0, 0, 0

# 5. Tentukan Risk Level
if total == 0:
    risk_level, risk_color, risk_msg = "DATA TIDAK TERSEDIA", "gray", "Tidak ada data historis."
elif pct_high > 0.5 or max_mag >= 5.5:
    risk_level, risk_color = "BAHAYA TINGGI 🔴", "red"
    risk_msg = f"Wilayah ini memiliki riwayat gempa berbahaya di bulan {current_month_name}."
elif avg_mag >= 4.0 or pct_high > 0.25:
    risk_level, risk_color = "WASPADA 🟠", "orange"
    risk_msg = "Ada aktivitas seismik signifikan. Tetap waspada."
else:
    risk_level, risk_color = "RELATIF AMAN 🟢", "green"
    risk_msg = "Aktivitas seismik relatif rendah berdasarkan histori."

# 6. Tampilkan
st.sidebar.markdown(f"""
**📊 Ringkasan Historis ({current_month_name})**
- Total Gempa: {total}
- Avg Mag: {avg_mag:.2f}
- Max Mag: {max_mag:.2f}
- High/Very High: {pct_high*100:.0f}%
""")

st.sidebar.markdown(f"""
<div style="background-color:{risk_color}; padding:10px; border-radius:5px; color:white; text-align:center;">
<strong>{risk_level}</strong><br>{risk_msg}
</div>
""", unsafe_allow_html=True)

# 7. Tampilkan Tips (Hanya jika risiko tinggi/waspada)
if risk_color in ["red", "orange"]:
    with st.sidebar.expander("💡 Tips Keselamatan", expanded=True):
        st.markdown("**Sebelum Gempa:**")
        for tip in TIPS_SEBELUM_GEMPA:
            st.markdown(f"- {tip}")
        st.markdown("**Saat Gempa:**")
        for tip in TIPS_SAAT_GEMPA:
            st.markdown(f"- {tip}")
        st.markdown("**Setelah Gempa:**")
        for tip in TIPS_SETELAH_GEMPA:
            st.markdown(f"- {tip}")

6. CATATAN TAMBAHAN
================================================================================

A. DATA BOUNDING BOX
   Koordinat wilayah di atas adalah PERKIRAAN kasar. Untuk akurasi lebih baik,
   gunakan data shapefile kabupaten/kota atau API geocoding seperti Nominatim.

B. ALTERNATIF: INPUT KOTA
   Jika ingin input kota spesifik (misal: "Semarang"), bisa buat dictionary:
   KOTA_COORDS = {
       "Semarang":   {"lat": -6.97, "lon": 110.42, "radius": 0.5},
       "Surabaya":   {"lat": -7.25, "lon": 112.75, "radius": 0.5},
       ...
   }
   Lalu filter dengan jarak euclidean atau haversine.

C. REFRESH OTOMATIS
   Jika ingin update bulan otomatis setiap hari, tambahkan:
   st.experimental_rerun() dengan scheduler atau cron.

D. INTEGRASI API BMKG (OPSIONAL)
   Untuk data real-time, bisa integrasikan dengan API BMKG:
   https://data.bmkg.go.id/DataMKG/TEWS/autogempa.json
